"use client";

import Link from "next/link";
import {
  useEffect,
  useMemo,
  useState,
} from "react";
import { BlockMath } from "react-katex";
import "katex/dist/katex.min.css";

import SiteHeader from "@/components/SiteHeader";
import {
  practiceTopics,
  questions,
  type QuestionOption,
} from "@/data/questions";
import { createClient } from "@/lib/supabase/client";

/* =========================
   类型
========================= */

type Mistake = {
  id: number;
  question_id: string;
  topic: string;

  title: string | null;
  question: string;
  formula: string | null;
  options: QuestionOption[] | null;

  user_answer: string;
  correct_answer: string;
  explanation: string | null;

  created_at: string;
};

type ResolvedQuestion = {
  id: string;
  topic: string;
  title: string;
  question: string;
  formula?: string;
  options: QuestionOption[];
  correctAnswer: string;
  explanation: string;
};

type FilterTopic =
  | "全部"
  | string;

/* =========================
   页面
========================= */

export default function MistakesPage() {
  const supabase = useMemo(
    () => createClient(),
    []
  );

  const [
    mistakes,
    setMistakes,
  ] = useState<Mistake[]>([]);

  const [
    loading,
    setLoading,
  ] = useState(true);

  const [
    loggedIn,
    setLoggedIn,
  ] = useState(false);

  const [
    userEmail,
    setUserEmail,
  ] = useState<
    string | null
  >(null);

  const [
    selectedTopic,
    setSelectedTopic,
  ] =
    useState<FilterTopic>(
      "全部"
    );

  const [
    activeRetryId,
    setActiveRetryId,
  ] = useState<
    number | null
  >(null);

  const [
    retryAnswer,
    setRetryAnswer,
  ] = useState<
    string | null
  >(null);

  const [
    retryResult,
    setRetryResult,
  ] = useState<
    | "correct"
    | "wrong"
    | null
  >(null);

  const [
    retrySaving,
    setRetrySaving,
  ] = useState(false);

  const [
    actionMessage,
    setActionMessage,
  ] = useState("");

  /* =========================
     读取错题
  ========================= */

  useEffect(() => {
    loadMistakes();
  }, []);

  async function loadMistakes() {
    setLoading(true);

    const {
      data: { user },
    } =
      await supabase.auth.getUser();

    if (!user) {
      setLoggedIn(false);
      setUserEmail(null);
      setMistakes([]);
      setLoading(false);
      return;
    }

    setLoggedIn(true);

    setUserEmail(
      user.email ?? null
    );

    const {
      data,
      error,
    } = await supabase
      .from("mistakes")
      .select(
        `
          id,
          question_id,
          topic,
          title,
          question,
          formula,
          options,
          user_answer,
          correct_answer,
          explanation,
          created_at
        `
      )
      .eq(
        "user_id",
        user.id
      )
      .order(
        "created_at",
        {
          ascending: false,
        }
      );

    if (error) {
      console.error(
        "读取错题失败：",
        error
      );

      setMistakes([]);
      setLoading(false);
      return;
    }

    setMistakes(
      (data ??
        []) as Mistake[]
    );

    setLoading(false);
  }

  /* =========================
     筛选
  ========================= */

  const filteredMistakes =
    selectedTopic ===
    "全部"
      ? mistakes
      : mistakes.filter(
          (mistake) =>
            normalizeTopic(
              mistake.topic
            ) ===
            selectedTopic
        );

  function topicCount(
    topic: string
  ) {
    return mistakes.filter(
      (mistake) =>
        normalizeTopic(
          mistake.topic
        ) === topic
    ).length;
  }

  /* =========================
     打开重新作答
  ========================= */

  function startRetry(
    mistakeId: number
  ) {
    if (
      activeRetryId ===
      mistakeId
    ) {
      setActiveRetryId(
        null
      );

      setRetryAnswer(
        null
      );

      setRetryResult(
        null
      );

      return;
    }

    setActiveRetryId(
      mistakeId
    );

    setRetryAnswer(null);

    setRetryResult(null);

    setActionMessage("");
  }

  /* =========================
     重新作答
  ========================= */

  async function submitRetry(
    mistake: Mistake,
    question: ResolvedQuestion
  ) {
    if (
      !retryAnswer ||
      retrySaving
    ) {
      return;
    }

    const {
      data: { user },
    } =
      await supabase.auth.getUser();

    if (!user) {
      setActionMessage(
        "登录状态已失效，请重新登录。"
      );

      return;
    }

    setRetrySaving(true);

    const correct =
      retryAnswer ===
      question.correctAnswer;

    const selectedOption =
      question.options.find(
        (option) =>
          option.key ===
          retryAnswer
      );

    const correctOption =
      question.options.find(
        (option) =>
          option.key ===
          question.correctAnswer
      );

    const selectedAnswerText =
      selectedOption
        ? `${selectedOption.key}. ${selectedOption.text}`
        : retryAnswer;

    const correctAnswerText =
      correctOption
        ? `${correctOption.key}. ${correctOption.text}`
        : question.correctAnswer;

    /* =========================
       记录重新作答
    ========================= */

    const {
      error:
        attemptError,
    } = await supabase
      .from(
        "practice_attempts"
      )
      .insert({
        user_id:
          user.id,

        question_id:
          question.id,

        topic:
          question.topic,

        selected_answer:
          selectedAnswerText,

        correct_answer:
          correctAnswerText,

        is_correct:
          correct,
      });

    if (attemptError) {
      console.error(
        "重新作答保存失败：",
        attemptError
      );

      setActionMessage(
        "重新作答记录保存失败，请稍后再试。"
      );

      setRetrySaving(
        false
      );

      return;
    }

    /* =========================
       仍然答错
    ========================= */

    if (!correct) {
      setRetryResult(
        "wrong"
      );

      setActionMessage(
        "这次还没有答对，题目继续保留在错题本中。"
      );

      setRetrySaving(
        false
      );

      return;
    }

    /* =========================
       答对 → 删除错题
    ========================= */

    const {
      error:
        deleteError,
    } = await supabase
      .from("mistakes")
      .delete()
      .eq(
        "id",
        mistake.id
      )
      .eq(
        "user_id",
        user.id
      );

    if (deleteError) {
      console.error(
        "删除错题失败：",
        deleteError
      );

      setActionMessage(
        "答案正确，但从错题本移除失败，请刷新后重试。"
      );

      setRetrySaving(
        false
      );

      return;
    }

    setMistakes(
      (prev) =>
        prev.filter(
          (item) =>
            item.id !==
            mistake.id
        )
    );

    setActiveRetryId(null);
    setRetryAnswer(null);
    setRetryResult(null);

    setActionMessage(
      "回答正确，这道题已经自动移出错题本，并计入新的练习记录。"
    );

    setRetrySaving(false);
  }

  /* =========================
     删除单题
  ========================= */

  async function deleteMistake(
    mistakeId: number
  ) {
    const {
      data: { user },
    } =
      await supabase.auth.getUser();

    if (!user) {
      return;
    }

    const confirmed =
      window.confirm(
        "确定要把这道题从错题本中移除吗？"
      );

    if (!confirmed) {
      return;
    }

    const { error } =
      await supabase
        .from("mistakes")
        .delete()
        .eq(
          "id",
          mistakeId
        )
        .eq(
          "user_id",
          user.id
        );

    if (error) {
      console.error(
        error
      );

      setActionMessage(
        "删除失败，请稍后再试。"
      );

      return;
    }

    setMistakes(
      (prev) =>
        prev.filter(
          (mistake) =>
            mistake.id !==
            mistakeId
        )
    );

    if (
      activeRetryId ===
      mistakeId
    ) {
      setActiveRetryId(
        null
      );

      setRetryAnswer(
        null
      );

      setRetryResult(
        null
      );
    }

    setActionMessage(
      "这道题已经移出错题本。"
    );
  }

  /* =========================
     清空错题本
  ========================= */

  async function clearAllMistakes() {
    if (
      mistakes.length === 0
    ) {
      return;
    }

    const confirmed =
      window.confirm(
        `确定要清空全部 ${mistakes.length} 道错题吗？此操作无法撤销。`
      );

    if (!confirmed) {
      return;
    }

    const {
      data: { user },
    } =
      await supabase.auth.getUser();

    if (!user) {
      return;
    }

    const { error } =
      await supabase
        .from("mistakes")
        .delete()
        .eq(
          "user_id",
          user.id
        );

    if (error) {
      console.error(
        error
      );

      setActionMessage(
        "清空错题本失败，请稍后再试。"
      );

      return;
    }

    setMistakes([]);

    setActiveRetryId(
      null
    );

    setRetryAnswer(null);

    setRetryResult(null);

    setSelectedTopic(
      "全部"
    );

    setActionMessage(
      "错题本已经清空。"
    );
  }

  /* =========================
     加载
  ========================= */

  if (loading) {
    return (
      <main className="min-h-screen bg-[#fafafa]">
        <SiteHeader />

        <div className="mx-auto max-w-6xl px-4 py-20 sm:px-6 lg:px-8">
          <div className="rounded-2xl border border-slate-200 bg-white p-10 text-center">
            <p className="text-sm text-slate-400">
              正在加载错题本...
            </p>
          </div>
        </div>
      </main>
    );
  }

  /* =========================
     未登录
  ========================= */

  if (!loggedIn) {
    return (
      <main className="min-h-screen bg-[#fafafa]">
        <SiteHeader />

        <div className="mx-auto max-w-6xl px-4 py-20 sm:px-6 lg:px-8">
          <div className="mx-auto max-w-xl rounded-2xl border border-slate-200 bg-white p-8 text-center sm:p-10">
            <p className="text-sm font-medium text-blue-600">
              我的错题
            </p>

            <h1 className="mt-3 text-2xl font-bold text-slate-900">
              登录后查看个人错题本
            </h1>

            <p className="mt-4 text-sm leading-7 text-slate-500">
              登录后，
              专题训练中答错的题目会自动保存到云端，
              可以跨设备查看和重新作答。
            </p>

            <Link
              href="/login"
              className="mt-6 inline-block rounded-lg bg-slate-900 px-5 py-2.5 text-sm font-medium text-white hover:bg-slate-700"
            >
              登录 / 注册
            </Link>
          </div>
        </div>
      </main>
    );
  }

  /* =========================
     页面
  ========================= */

  return (
    <main className="min-h-screen bg-[#fafafa]">
      <SiteHeader />

      <div className="mx-auto max-w-6xl px-4 pb-20 pt-10 sm:px-6 lg:px-8">
        {/* 面包屑 */}
        <div className="mb-10 flex flex-wrap items-center gap-2 text-sm text-slate-400">
          <Link
            href="/"
            className="hover:text-slate-700"
          >
            首页
          </Link>

          <span>/</span>

          <span className="text-slate-600">
            我的错题
          </span>
        </div>

        {/* 标题 */}
        <section className="border-b border-slate-200 pb-9">
          <div className="flex flex-col justify-between gap-6 sm:flex-row sm:items-end">
            <div>
              <p className="text-sm font-medium text-blue-600">
                MISTAKE BOOK
              </p>

              <h1 className="mt-3 text-3xl font-bold tracking-tight text-slate-900 sm:text-4xl">
                我的错题
              </h1>

              <p className="mt-4 max-w-2xl text-base leading-8 text-slate-500">
                集中复习专题训练中答错的题目。
                现在会完整保存题目、
                公式、选项和解析。
                重新作答正确后自动移出错题本。
              </p>
            </div>

            <div className="rounded-xl border border-slate-200 bg-white px-5 py-4">
              <p className="text-xs text-slate-400">
                当前学生
              </p>

              <p className="mt-2 max-w-[220px] truncate text-sm font-medium text-slate-800">
                {userEmail}
              </p>
            </div>
          </div>
        </section>

        {/* 统计 */}
        <section className="grid gap-3 py-8 sm:grid-cols-3">
          <StatCard
            label="错题总数"
            value={String(
              mistakes.length
            )}
            suffix="题"
          />

          <StatCard
            label="涉及专题"
            value={String(
              practiceTopics.filter(
                (topic) =>
                  topicCount(
                    topic
                  ) > 0
              ).length
            )}
            suffix="个"
          />

          <StatCard
            label="当前筛选"
            value={
              selectedTopic ===
              "全部"
                ? "全部"
                : selectedTopic
            }
          />
        </section>

        {/* 全局提示 */}
        {actionMessage && (
          <div className="mb-6 flex items-start justify-between gap-4 rounded-xl border border-blue-100 bg-blue-50/50 px-5 py-4">
            <p className="text-sm leading-6 text-slate-600">
              {
                actionMessage
              }
            </p>

            <button
              type="button"
              onClick={() =>
                setActionMessage(
                  ""
                )
              }
              className="shrink-0 text-sm text-slate-400 hover:text-slate-700"
            >
              ×
            </button>
          </div>
        )}

        <div className="grid gap-8 lg:grid-cols-[250px_1fr]">
          {/* =====================
              左侧专题
          ====================== */}

          <aside>
            <div className="rounded-2xl border border-slate-200 bg-white p-4">
              <p className="px-2 text-xs font-medium text-slate-400">
                按专题筛选
              </p>

              <div className="mt-3 space-y-1">
                <FilterButton
                  label="全部错题"
                  count={
                    mistakes.length
                  }
                  active={
                    selectedTopic ===
                    "全部"
                  }
                  onClick={() =>
                    setSelectedTopic(
                      "全部"
                    )
                  }
                />

                {practiceTopics.map(
                  (
                    topic,
                    index
                  ) => (
                    <FilterButton
                      key={
                        topic
                      }
                      label={`${String(
                        index +
                          1
                      ).padStart(
                        2,
                        "0"
                      )} ${topic}`}
                      count={topicCount(
                        topic
                      )}
                      active={
                        selectedTopic ===
                        topic
                      }
                      onClick={() =>
                        setSelectedTopic(
                          topic
                        )
                      }
                    />
                  )
                )}
              </div>
            </div>

            <div className="mt-4 rounded-xl border border-slate-200 bg-white p-5">
              <p className="text-xs text-slate-400">
                错题规则
              </p>

              <p className="mt-2 text-sm leading-6 text-slate-500">
                专题训练答错后自动加入这里。
                新错题会保存完整题目快照，
                在错题本重新答对后自动移除。
              </p>

              <Link
                href="/practice"
                className="mt-4 inline-block text-sm font-medium text-slate-800"
              >
                去专题训练 →
              </Link>
            </div>

            {mistakes.length >
              0 && (
              <button
                type="button"
                onClick={
                  clearAllMistakes
                }
                className="mt-4 w-full rounded-xl border border-slate-200 bg-white px-4 py-3 text-sm text-slate-500 transition hover:border-red-200 hover:bg-red-50 hover:text-red-500"
              >
                清空全部错题
              </button>
            )}
          </aside>

          {/* =====================
              右侧错题
          ====================== */}

          <section>
            <div className="mb-5 flex flex-wrap items-end justify-between gap-4">
              <div>
                <h2 className="text-xl font-bold text-slate-900">
                  {selectedTopic ===
                  "全部"
                    ? "全部错题"
                    : selectedTopic}
                </h2>

                <p className="mt-1 text-sm text-slate-500">
                  当前共{" "}
                  {
                    filteredMistakes.length
                  }{" "}
                  道
                </p>
              </div>

              <Link
                href="/report"
                className="text-sm font-medium text-slate-700"
              >
                查看学习报告 →
              </Link>
            </div>

            {filteredMistakes.length ===
            0 ? (
              <EmptyMistakes
                hasAnyMistakes={
                  mistakes.length >
                  0
                }
              />
            ) : (
              <div className="space-y-5">
                {filteredMistakes.map(
                  (
                    mistake,
                    index
                  ) => {
                    const question =
                      resolveQuestion(
                        mistake
                      );

                    const retryOpen =
                      activeRetryId ===
                      mistake.id;

                    return (
                      <MistakeCard
                        key={
                          mistake.id
                        }
                        number={
                          index +
                          1
                        }
                        mistake={
                          mistake
                        }
                        question={
                          question
                        }
                        retryOpen={
                          retryOpen
                        }
                        retryAnswer={
                          retryAnswer
                        }
                        retryResult={
                          retryResult
                        }
                        retrySaving={
                          retrySaving
                        }
                        onStartRetry={() =>
                          startRetry(
                            mistake.id
                          )
                        }
                        onSelectAnswer={
                          setRetryAnswer
                        }
                        onSubmitRetry={() => {
                          if (
                            question
                          ) {
                            submitRetry(
                              mistake,
                              question
                            );
                          }
                        }}
                        onDelete={() =>
                          deleteMistake(
                            mistake.id
                          )
                        }
                      />
                    );
                  }
                )}
              </div>
            )}
          </section>
        </div>
      </div>
    </main>
  );
}

/* =========================
   错题卡
========================= */

function MistakeCard({
  number,
  mistake,
  question,
  retryOpen,
  retryAnswer,
  retryResult,
  retrySaving,
  onStartRetry,
  onSelectAnswer,
  onSubmitRetry,
  onDelete,
}: {
  number: number;
  mistake: Mistake;

  question:
    | ResolvedQuestion
    | null;

  retryOpen: boolean;

  retryAnswer:
    | string
    | null;

  retryResult:
    | "correct"
    | "wrong"
    | null;

  retrySaving: boolean;

  onStartRetry:
    () => void;

  onSelectAnswer: (
    answer: string
  ) => void;

  onSubmitRetry:
    () => void;

  onDelete:
    () => void;
}) {
  const originalAnswer =
    displayAnswer(
      mistake.user_answer,
      question
    );

  const correctAnswer =
    displayAnswer(
      mistake.correct_answer,
      question
    );

  const originalKey =
    extractAnswerKey(
      mistake.user_answer
    );

  const correctKey =
    question
      ? question.correctAnswer
      : extractAnswerKey(
          mistake.correct_answer
        );

  return (
    <article className="rounded-2xl border border-slate-200 bg-white p-6 sm:p-7">
      {/* 顶部 */}
      <div className="flex flex-wrap items-start justify-between gap-4">
        <div>
          <div className="flex flex-wrap items-center gap-2">
            <span className="rounded-full bg-red-50 px-2.5 py-1 text-xs font-medium text-red-500">
              错题{" "}
              {String(
                number
              ).padStart(
                2,
                "0"
              )}
            </span>

            <span className="text-xs text-slate-400">
              {normalizeTopic(
                mistake.topic
              )}
            </span>

            {hasSnapshot(
              mistake
            ) && (
              <span className="rounded-full bg-emerald-50 px-2 py-0.5 text-[11px] font-medium text-emerald-600">
                完整记录
              </span>
            )}
          </div>

          <h3 className="mt-4 text-lg font-semibold text-slate-900">
            {question?.title ??
              mistake.title ??
              "历史错题"}
          </h3>
        </div>

        <p className="text-xs text-slate-400">
          {formatDate(
            mistake.created_at
          )}
        </p>
      </div>

      {/* =====================
          题干
      ====================== */}

      <div className="mt-6">
        <p className="text-sm leading-7 text-slate-700">
          {question?.question ??
            mistake.question}
        </p>

        {(question?.formula ||
          mistake.formula) && (
          <div className="my-6 overflow-x-auto rounded-xl bg-slate-50 px-4 py-5 text-center">
            <BlockMath
              math={
                question?.formula ??
                mistake.formula ??
                ""
              }
            />
          </div>
        )}
      </div>

      {/* =====================
          完整选项
      ====================== */}

      {question &&
      question.options.length >
        0 ? (
        <div className="mt-6 grid gap-3">
          {question.options.map(
            (option) => {
              const wasSelected =
                option.key ===
                originalKey;

              const isAnswer =
                option.key ===
                correctKey;

              return (
                <div
                  key={
                    option.key
                  }
                  className={`flex items-start gap-3 rounded-xl border px-4 py-4 text-sm leading-6 ${
                    isAnswer
                      ? "border-emerald-200 bg-emerald-50/60 text-emerald-700"
                      : wasSelected
                        ? "border-red-200 bg-red-50/60 text-red-600"
                        : "border-slate-200 bg-white text-slate-600"
                  }`}
                >
                  <span
                    className={`flex h-7 w-7 shrink-0 items-center justify-center rounded-md border text-xs font-semibold ${
                      isAnswer
                        ? "border-emerald-300 bg-white text-emerald-600"
                        : wasSelected
                          ? "border-red-300 bg-white text-red-500"
                          : "border-slate-200 bg-slate-50 text-slate-500"
                    }`}
                  >
                    {
                      option.key
                    }
                  </span>

                  <span className="pt-0.5">
                    {
                      option.text
                    }
                  </span>

                  {isAnswer && (
                    <span className="ml-auto shrink-0 text-xs font-medium text-emerald-600">
                      正确答案
                    </span>
                  )}

                  {!isAnswer &&
                    wasSelected && (
                      <span className="ml-auto shrink-0 text-xs font-medium text-red-500">
                        当时选择
                      </span>
                    )}
                </div>
              );
            }
          )}
        </div>
      ) : (
        <div className="mt-6 rounded-xl border border-amber-100 bg-amber-50/50 p-4">
          <p className="text-sm font-medium text-amber-700">
            历史错题缺少完整题目数据
          </p>

          <p className="mt-2 text-sm leading-6 text-slate-500">
            这道题保存时还没有记录公式和
            A/B/C/D 选项，
            并且当前题库中已经找不到对应的
            question ID，
            因此无法自动恢复原始选项。
          </p>
        </div>
      )}

      {/* =====================
          原答题结果
      ====================== */}

      <div className="mt-6 grid gap-3 sm:grid-cols-2">
        <div className="rounded-xl bg-red-50/60 p-4">
          <p className="text-xs font-medium text-red-500">
            当时的答案
          </p>

          <p className="mt-2 text-sm leading-6 text-slate-700">
            {
              originalAnswer
            }
          </p>
        </div>

        <div className="rounded-xl bg-emerald-50/60 p-4">
          <p className="text-xs font-medium text-emerald-600">
            正确答案
          </p>

          <p className="mt-2 text-sm leading-6 text-slate-700">
            {
              correctAnswer
            }
          </p>
        </div>
      </div>

      {/* =====================
          解析
      ====================== */}

      {(question?.explanation ||
        mistake.explanation) && (
        <div className="mt-5 rounded-xl bg-slate-50 p-4">
          <p className="text-xs font-medium text-slate-400">
            题目解析
          </p>

          <p className="mt-2 text-sm leading-7 text-slate-600">
            {question?.explanation ??
              mistake.explanation}
          </p>
        </div>
      )}

      {/* =====================
          操作
      ====================== */}

      <div className="mt-6 flex flex-wrap gap-3">
        {question ? (
          <button
            type="button"
            onClick={
              onStartRetry
            }
            className="rounded-lg bg-slate-900 px-4 py-2.5 text-sm font-medium text-white hover:bg-slate-700"
          >
            {retryOpen
              ? "收起重新作答"
              : "重新作答"}
          </button>
        ) : (
          <div className="rounded-lg bg-slate-100 px-4 py-2.5 text-sm text-slate-400">
            该历史题暂无完整选项
          </div>
        )}

        <button
          type="button"
          onClick={onDelete}
          className="rounded-lg border border-slate-200 px-4 py-2.5 text-sm text-slate-500 hover:border-red-200 hover:bg-red-50 hover:text-red-500"
        >
          移出错题本
        </button>
      </div>

      {/* =====================
          重新作答
      ====================== */}

      {retryOpen &&
        question && (
          <div className="mt-6 border-t border-slate-100 pt-6">
            <div>
              <p className="text-sm font-semibold text-slate-900">
                再做一次
              </p>

              <p className="mt-1 text-xs text-slate-400">
                这次答对后会自动移出错题本
              </p>
            </div>

            <div className="mt-5 grid gap-3">
              {question.options.map(
                (
                  option
                ) => {
                  const chosen =
                    retryAnswer ===
                    option.key;

                  const showCorrect =
                    retryResult ===
                      "wrong" &&
                    option.key ===
                      question.correctAnswer;

                  const showWrong =
                    retryResult ===
                      "wrong" &&
                    chosen &&
                    option.key !==
                      question.correctAnswer;

                  return (
                    <button
                      key={
                        option.key
                      }
                      type="button"
                      disabled={
                        retryResult !==
                          null ||
                        retrySaving
                      }
                      onClick={() =>
                        onSelectAnswer(
                          option.key
                        )
                      }
                      className={`flex items-start gap-3 rounded-xl border px-4 py-4 text-left text-sm leading-6 transition ${
                        showCorrect
                          ? "border-emerald-300 bg-emerald-50 text-emerald-700"
                          : showWrong
                            ? "border-red-300 bg-red-50 text-red-600"
                            : chosen
                              ? "border-slate-500 bg-slate-50 text-slate-900"
                              : "border-slate-200 text-slate-600 hover:border-slate-300 hover:bg-slate-50"
                      }`}
                    >
                      <span
                        className={`flex h-7 w-7 shrink-0 items-center justify-center rounded-full border text-xs font-semibold ${
                          showCorrect
                            ? "border-emerald-300 bg-white text-emerald-600"
                            : showWrong
                              ? "border-red-300 bg-white text-red-500"
                              : chosen
                                ? "border-slate-800 bg-slate-900 text-white"
                                : "border-slate-200 bg-white text-slate-500"
                        }`}
                      >
                        {
                          option.key
                        }
                      </span>

                      <span className="pt-0.5">
                        {
                          option.text
                        }
                      </span>
                    </button>
                  );
                }
              )}
            </div>

            {retryResult ===
              "wrong" && (
              <div className="mt-5 rounded-xl border border-red-100 bg-red-50/50 p-4">
                <p className="text-sm font-semibold text-red-600">
                  这次还没有答对
                </p>

                <p className="mt-2 text-sm leading-7 text-slate-600">
                  正确答案是{" "}
                  {
                    question.correctAnswer
                  }
                  。
                  这道题会继续保留在错题本中。
                </p>

                <p className="mt-3 text-sm leading-7 text-slate-600">
                  {
                    question.explanation
                  }
                </p>
              </div>
            )}

            {retryResult ===
              null && (
              <button
                type="button"
                onClick={
                  onSubmitRetry
                }
                disabled={
                  !retryAnswer ||
                  retrySaving
                }
                className="mt-5 rounded-lg bg-slate-900 px-5 py-2.5 text-sm font-medium text-white hover:bg-slate-700 disabled:cursor-not-allowed disabled:opacity-30"
              >
                {retrySaving
                  ? "正在提交..."
                  : "提交答案"}
              </button>
            )}

            {retryResult ===
              "wrong" && (
              <button
                type="button"
                onClick={
                  onStartRetry
                }
                className="mt-4 text-sm text-slate-500 hover:text-slate-900"
              >
                收起
              </button>
            )}
          </div>
        )}
    </article>
  );
}

/* =========================
   解析错题完整数据
========================= */

function resolveQuestion(
  mistake: Mistake
): ResolvedQuestion | null {
  /*
    第一优先级：
    Supabase 保存的完整题目快照

    这样即使以后 questions.ts
    修改，旧错题仍保持原样。
  */

  const snapshotOptions =
    normalizeOptions(
      mistake.options
    );

  if (
    snapshotOptions.length >
    0
  ) {
    const correctKey =
      extractAnswerKey(
        mistake.correct_answer
      ) ??
      mistake.correct_answer;

    return {
      id:
        mistake.question_id,

      topic:
        normalizeTopic(
          mistake.topic
        ),

      title:
        mistake.title ??
        "错题",

      question:
        mistake.question,

      formula:
        mistake.formula ??
        undefined,

      options:
        snapshotOptions,

      correctAnswer:
        correctKey,

      explanation:
        mistake.explanation ??
        "",
    };
  }

  /*
    第二优先级：
    老错题没有快照，
    就从当前题库通过 ID 查找。

    如果 question_id 仍存在，
    老错题仍然可以恢复完整显示。
  */

  const currentQuestion =
    questions.find(
      (item) =>
        item.id ===
        mistake.question_id
    );

  if (
    currentQuestion
  ) {
    return {
      id:
        currentQuestion.id,

      topic:
        currentQuestion.topic,

      title:
        currentQuestion.title,

      question:
        currentQuestion.question,

      formula:
        currentQuestion.formula,

      options:
        currentQuestion.options,

      correctAnswer:
        currentQuestion.correctAnswer,

      explanation:
        currentQuestion.explanation,
    };
  }

  return null;
}

/* =========================
   JSONB options 安全处理
========================= */

function normalizeOptions(
  options:
    | QuestionOption[]
    | null
): QuestionOption[] {
  if (
    !Array.isArray(
      options
    )
  ) {
    return [];
  }

  return options.filter(
    (
      option
    ): option is QuestionOption =>
      Boolean(
        option &&
          typeof option.key ===
            "string" &&
          typeof option.text ===
            "string"
      )
  );
}

/* =========================
   判断是否为新完整快照
========================= */

function hasSnapshot(
  mistake: Mistake
) {
  return (
    Boolean(
      mistake.title
    ) &&
    normalizeOptions(
      mistake.options
    ).length > 0
  );
}

/* =========================
   答案格式
========================= */

function extractAnswerKey(
  value:
    | string
    | null
    | undefined
) {
  if (!value) {
    return null;
  }

  const trimmed =
    value
      .trim()
      .toUpperCase();

  const direct =
    trimmed.match(
      /^([A-Z])$/
    );

  if (direct) {
    return direct[1];
  }

  const withText =
    trimmed.match(
      /^([A-Z])[\.\s、:]/
    );

  if (withText) {
    return withText[1];
  }

  return null;
}

function displayAnswer(
  storedAnswer: string,
  question:
    | ResolvedQuestion
    | null
) {
  const key =
    extractAnswerKey(
      storedAnswer
    );

  if (
    key &&
    question
  ) {
    const option =
      question.options.find(
        (item) =>
          item.key ===
          key
      );

    if (option) {
      return `${key}. ${option.text}`;
    }
  }

  return storedAnswer;
}

function normalizeTopic(
  topic: string
): string {
  if (
    topicAliases[topic]
  ) {
    return topicAliases[
      topic
    ];
  }

  if (
    topic.includes(
      "函数的概念"
    )
  ) {
    return "函数的概念";
  }

  if (
    topic.includes(
      "定义域"
    ) ||
    topic.includes(
      "值域"
    )
  ) {
    return "定义域与值域";
  }

  if (
    topic.includes(
      "单调"
    )
  ) {
    return "函数的单调性";
  }

  if (
    topic.includes(
      "奇偶"
    )
  ) {
    return "函数的奇偶性";
  }

  if (
    topic.includes(
      "最大值"
    ) ||
    topic.includes(
      "最小值"
    ) ||
    topic.includes(
      "最值"
    )
  ) {
    return "函数的最大值与最小值";
  }

  return topic;
}
